/**
 * Required Problem 6 Interface:
 * choose_move(board, player) -> column (0 to 6)
 *
 * @param {number[][]} board 6x7 grid (0=empty, 1=player 1, 2=player 2)
 * @param {number} player 1 or 2
 * @returns {number} chosen column index 0 to 6
 */
function choose_move(board, player) {
    const ROWS = 6, COLS = 7, EMPTY = 0;
    const opp = player === 1 ? 2 : 1;
    const ORDER = [3, 2, 4, 1, 5, 0, 6];

    function validCols() {
        return ORDER.filter(c => board[0][c] === EMPTY);
    }

    function openRow(c) {
        for (let r = ROWS - 1; r >= 0; r--) {
            if (board[r][c] === EMPTY) return r;
        }
        return -1;
    }

    function isWin(p) {
        // Horiz
        for (let r=0; r<ROWS; r++)
            for (let c=0; c<COLS-3; c++)
                if (board[r][c]===p && board[r][c+1]===p && board[r][c+2]===p && board[r][c+3]===p) return true;
        // Vert
        for (let r=0; r<ROWS-3; r++)
            for (let c=0; c<COLS; c++)
                if (board[r][c]===p && board[r+1][c]===p && board[r+2][c]===p && board[r+3][c]===p) return true;
        // Diag Up
        for (let r=3; r<ROWS; r++)
            for (let c=0; c<COLS-3; c++)
                if (board[r][c]===p && board[r-1][c+1]===p && board[r-2][c+2]===p && board[r-3][c+3]===p) return true;
        // Diag Down
        for (let r=0; r<ROWS-3; r++)
            for (let c=0; c<COLS-3; c++)
                if (board[r][c]===p && board[r+1][c+1]===p && board[r+2][c+2]===p && board[r+3][c+3]===p) return true;
        return false;
    }

    const valids = validCols();
    if (!valids.length) return 0;

    // 1. Immediate Win
    for (const c of valids) {
        const r = openRow(c);
        board[r][c] = player;
        if (isWin(player)) { board[r][c] = EMPTY; return c; }
        board[r][c] = EMPTY;
    }

    // 2. Immediate Block
    for (const c of valids) {
        const r = openRow(c);
        board[r][c] = opp;
        if (isWin(opp)) { board[r][c] = EMPTY; return c; }
        board[r][c] = EMPTY;
    }

    // 3. Minimax with Alpha-Beta Pruning (Depth 4)
    function scoreWindow(w) {
        let p=0, e=0, o=0;
        for (const v of w) { if (v===player) p++; else if (v===EMPTY) e++; else if (v===opp) o++; }
        if (p===4) return 100000;
        if (p===3 && e===1) return 120;
        if (p===2 && e===2) return 12;
        if (o===3 && e===1) return -100;
        if (o===2 && e===2) return -10;
        return 0;
    }

    function evaluate() {
        let s = 0;
        for (let r=0; r<ROWS; r++) if (board[r][3] === player) s += 6;
        for (let r=0; r<ROWS; r++)
            for (let c=0; c<COLS-3; c++) s += scoreWindow([board[r][c], board[r][c+1], board[r][c+2], board[r][c+3]]);
        for (let c=0; c<COLS; c++)
            for (let r=0; r<ROWS-3; r++) s += scoreWindow([board[r][c], board[r+1][c], board[r+2][c], board[r+3][c]]);
        for (let r=3; r<ROWS; r++)
            for (let c=0; c<COLS-3; c++) s += scoreWindow([board[r][c], board[r-1][c+1], board[r-2][c+2], board[r-3][c+3]]);
        for (let r=0; r<ROWS-3; r++)
            for (let c=0; c<COLS-3; c++) s += scoreWindow([board[r][c], board[r+1][c+1], board[r+2][c+2], board[r+3][c+3]]);
        return s;
    }

    function minimax(depth, alpha, beta, isMax) {
        const legal = validCols();
        if (isWin(player)) return { score: 10000000 + depth };
        if (isWin(opp)) return { score: -10000000 - depth };
        if (!legal.length || depth === 0) return { score: evaluate() };

        if (isMax) {
            let maxScore = -Infinity, best = legal[0];
            for (const c of legal) {
                const r = openRow(c);
                board[r][c] = player;
                const res = minimax(depth - 1, alpha, beta, false);
                board[r][c] = EMPTY;
                if (res.score > maxScore) { maxScore = res.score; best = c; }
                alpha = Math.max(alpha, maxScore);
                if (alpha >= beta) break;
            }
            return { col: best, score: maxScore };
        } else {
            let minScore = Infinity, best = legal[0];
            for (const c of legal) {
                const r = openRow(c);
                board[r][c] = opp;
                const res = minimax(depth - 1, alpha, beta, true);
                board[r][c] = EMPTY;
                if (res.score < minScore) { minScore = res.score; best = c; }
                beta = Math.min(beta, minScore);
                if (alpha >= beta) break;
            }
            return { col: best, score: minScore };
        }
    }

    return minimax(4, -Infinity, Infinity, true).col ?? valids[0];
}

import math

ROWS, COLS = 6, 7
COLUMN_ORDER = [3, 2, 4, 1, 5, 0, 6]

def is_valid_location(board, col):
    return board[0][col] == 0

def get_valid_locations(board):
    return [c for c in COLUMN_ORDER if is_valid_location(board, c)]

def get_next_open_row(board, col):
    for r in range(ROWS - 1, -1, -1):
        if board[r][col] == 0:
            return r
    return -1

def check_win(board, piece):
    for r in range(ROWS):
        for c in range(COLS - 3):
            if all(board[r][c+i] == piece for i in range(4)): return True
    for c in range(COLS):
        for r in range(ROWS - 3):
            if all(board[r+i][c] == piece for i in range(4)): return True
    for r in range(3, ROWS):
        for c in range(COLS - 3):
            if all(board[r-i][c+i] == piece for i in range(4)): return True
    for r in range(ROWS - 3):
        for c in range(COLS - 3):
            if all(board[r+i][c+i] == piece for i in range(4)): return True
    return False

def score_position(board, piece):
    opp = 1 if piece == 2 else 2
    score = sum(6 for r in range(ROWS) if board[r][3] == piece)
    def eval_window(w):
        p, e, o = w.count(piece), w.count(0), w.count(opp)
        if p == 4: return 100000
        if p == 3 and e == 1: return 120
        if p == 2 and e == 2: return 12
        if o == 3 and e == 1: return -100
        if o == 2 and e == 2: return -10
        return 0
    for r in range(ROWS):
        for c in range(COLS - 3): score += eval_window(board[r][c:c+4])
    for c in range(COLS):
        for r in range(ROWS - 3): score += eval_window([board[r+i][c] for i in range(4)])
    for r in range(3, ROWS):
        for c in range(COLS - 3): score += eval_window([board[r-i][c+i] for i in range(4)])
    for r in range(ROWS - 3):
        for c in range(COLS - 3): score += eval_window([board[r+i][c+i] for i in range(4)])
    return score

def choose_move(board, player, depth=4):
    """Returns column 0 to 6."""
    opp = 1 if player == 2 else 2
    valid_cols = get_valid_locations(board)
    if not valid_cols: return 0

    # 1. Immediate Win
    for col in valid_cols:
        r = get_next_open_row(board, col)
        board[r][col] = player
        if check_win(board, player):
            board[r][col] = 0; return col
        board[r][col] = 0

    # 2. Immediate Block
    for col in valid_cols:
        r = get_next_open_row(board, col)
        board[r][col] = opp
        if check_win(board, opp):
            board[r][col] = 0; return col
        board[r][col] = 0

    # 3. Minimax with Alpha-Beta
    def minimax(d, alpha, beta, is_max):
        legal = get_valid_locations(board)
        if check_win(board, player): return None, 10000000 + d
        if check_win(board, opp): return None, -10000000 - d
        if not legal or d == 0: return None, score_position(board, player)

        if is_max:
            val, best_c = -math.inf, legal[0]
            for c in legal:
                r = get_next_open_row(board, c)
                board[r][c] = player
                _, score = minimax(d - 1, alpha, beta, False)
                board[r][c] = 0
                if score > val: val, best_c = score, c
                alpha = max(alpha, val)
                if alpha >= beta: break
            return best_c, val
        else:
            val, best_c = math.inf, legal[0]
            for c in legal:
                r = get_next_open_row(board, c)
                board[r][c] = opp
                _, score = minimax(d - 1, alpha, beta, True)
                board[r][c] = 0
                if score < val: val, best_c = score, c
                beta = min(beta, val)
                if alpha >= beta: break
            return best_c, val

    col, _ = minimax(depth, -math.inf, math.inf, True)
    return col if col is not None else valid_cols[0]
